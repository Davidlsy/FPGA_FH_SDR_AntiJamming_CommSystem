1. 总体架构与时钟域

整个 eth_mac 由 6 个模块组成，数据流是单向的（RX 从左到右、TX 从右到左），MDIO 是一条独立配置面。

3. mdio_ctrl：MDIO 主控与 PHY 配置

RGMII 的 2ns 偏斜、速率、双工都靠 MDIO 写 PHY 寄存器来配，这是上电后必须第一步完成的事，否则后面全是时序幻觉。

3.1 MDIO 帧格式（IEEE 802.3 Clause 22）

字段位数值说明
Preamble32全 1前导，同步用
ST201帧起始
OP201=写 / 10=读操作码
PHYAD50~31PHY 地址（通常 0x00~0x07）
REGAD50~31寄存器地址
TA2写=10，读=Z0读时首 bit 由 PHY 驱动
DATA16—数据，MSB 先

合计 64 bit。MDC 用 50MHz 分频到 2.5MHz（分频 20）。读操作在 MDC 上升沿采样 mdio。

3.2 完整实现

module mdio_ctrl (
    input  wire        clk,          // 50MHz
    input  wire        rst_n,
    output wire        mdc,
    inout  wire        mdio,
    input  wire        start,        // 单拍脉冲：启动一次操作
    input  wire        op,           // 0=写 1=读
    input  wire [4:0]  phy_addr,
    input  wire [4:0]  reg_addr,
    input  wire [15:0] wr_data,
    output reg  [15:0] rd_data,
    output reg         busy,
    output reg         done
);
    // ---- MDC 生成：50MHz -> 2.5MHz ----
    reg [4:0] div_cnt;
    reg       mdc_r;
    reg       mdc_tick;      // MDC 下降沿脉冲，推进状态机
    always @(posedge clk or negedge rst_n) begin
        if (!rst_n) begin div_cnt <= 0; mdc_r <= 0; mdc_tick <= 0; end
        else begin
            mdc_tick <= 0;
            if (div_cnt == 5'd9) begin
                div_cnt <= 0; mdc_r <= ~mdc_r;
                if (mdc_r) mdc_tick <= 1;    // 下降沿
            end else
                div_cnt <= div_cnt + 1;
        end
    end
    assign mdc = mdc_r;

    // ---- 帧位序列（64 bit）----
    // bit 0..31  preamble=1
    // bit 32..33 ST=01
    // bit 34..35 OP={op,~op}
    // bit 36..40 PHYAD[4:0]
    // bit 41..45 REGAD[4:0]
    // bit 46..47 TA（读时 bit46 释放总线）
    // bit 48..63 DATA[15:0]
    reg [6:0] bit_cnt;
    reg       running;
    reg       mdio_oe;
    reg       mdio_out;
    reg       sampling;

    always @(posedge clk or negedge rst_n) begin
        if (!rst_n) begin
            running <= 0; bit_cnt <= 0; mdio_oe <= 0; mdio_out <= 0;
            sampling <= 0; busy <= 0; done <= 0; rd_data <= 0;
        end else begin
            done <= 0;
            if (start && !running) begin
                running <= 1; bit_cnt <= 0; busy <= 1;
                mdio_oe <= 1; mdio_out <= 1;        // preamble 首位
            end else if (running && mdc_tick) begin
                if (bit_cnt == 7'd63) begin
                    running <= 0; busy <= 0; done <= 1; mdio_oe <= 0;
                end else begin
                    bit_cnt <= bit_cnt + 1;
                    mdio_out <= next_bit(bit_cnt + 1);
                    // 读操作：TA 第 2 拍后、DATA 阶段释放总线并采样
                    if (op && (bit_cnt + 1) >= 7'd47) begin
                        mdio_oe <= 0;
                        sampling <= 1;
                    end
                end
            end
            if (sampling) rd_data <= {rd_data[14:0], mdio};
        end
    end

    // 计算第 n 拍要驱动的 bit（n 从 1 起）
    function next_bit;
        input [6:0] n;
        begin
            casez (n)
                7'd32:    next_bit = 0;                       // ST[0]
                7'd33:    next_bit = 1;                       // ST[1]
                7'd34:    next_bit = op;                      // OP[1]
                7'd35:    next_bit = ~op;                     // OP[0]
                7'd36:    next_bit = phy_addr[4];
                7'd37:    next_bit = phy_addr[3];
                7'd38:    next_bit = phy_addr[2];
                7'd39:    next_bit = phy_addr[1];
                7'd40:    next_bit = phy_addr[0];
                7'd41:    next_bit = reg_addr[4];
                7'd42:    next_bit = reg_addr[3];
                7'd43:    next_bit = reg_addr[2];
                7'd44:    next_bit = reg_addr[1];
                7'd45:    next_bit = reg_addr[0];
                7'd46:    next_bit = 1;                       // TA[0]（写）
                7'd47:    next_bit = 0;                       // TA[1]（写=0，读=Z）
                7'd48:    next_bit = wr_data[15];
                7'd49:    next_bit = wr_data[14];
                7'd50:    next_bit = wr_data[13];
                7'd51:    next_bit = wr_data[12];
                7'd52:    next_bit = wr_data[11];
                7'd53:    next_bit = wr_data[10];
                7'd54:    next_bit = wr_data[9];
                7'd55:    next_bit = wr_data[8];
                7'd56:    next_bit = wr_data[7];
                7'd57:    next_bit = wr_data[6];
                7'd58:    next_bit = wr_data[5];
                7'd59:    next_bit = wr_data[4];
                7'd60:    next_bit = wr_data[3];
                7'd61:    next_bit = wr_data[2];
                7'd62:    next_bit = wr_data[1];
                7'd63:    next_bit = wr_data[0];
                default: next_bit = 1;                       // preamble
            endcase
        end
    endfunction

    assign mdio = mdio_oe ? mdio_out : 1'bz;
endmodule

读采样时序：上面用 sampling 标志在 MDC 上升沿前采集 mdio，实际应确保采样点落在数据窗中心。更稳妥的做法是用 mdc_tick（下降沿）推进、在下一个上升沿采样；上板后若读回不稳定，用逻辑分析仪对一次 MDC/MDIO 波形即可校准。

3.3 PHY 上电初始化序列

一个简单状态机依次做：复位 → 开 RGMII delay → 自协商 → 读状态。寄存器地址因 PHY 芯片而异（RTL8211 / YT8531 / 88E1512 不同），以数据手册为准：

// 初始化状态机（示意，寄存器地址以实际 PHY 为准）
case (state)
    S_RST:    begin mdio_wr(phy, 0, 16'h8000); state <= S_WAIT_RST; end  // 软复位
    S_WAIT_RST: if (poll(phy,0,bit15)==0) state <= S_CFG_DELAY;
    S_CFG_DELAY: begin
        mdio_wr(phy, RGMII_DELAY_REG, RX_DELAY_2NS | TX_DELAY_2NS); // 关键
        state <= S_NEGO;
    end
    S_NEGO:   begin mdio_wr(phy, 0, 16'h1140); state <= S_WAIT_LINK; end // 千兆全双工自协商
    S_WAIT_LINK: if (poll(phy,1,bit2)==1) begin link_up <= 1; state <= S_DONE; end
endcase

务必先做：确认 PHY 的 RGMII delay 寄存器位（RX delay 通常必开，TX delay 与 FPGA 侧 90° 相移二选一）。这一步没做对，ping 会时通时断、吞吐上不去，且极难从数据层排查。

4. rgmii_if：RGMII 收发接口

把 DDR 4bit 转成单沿 8bit GMII 语义。高云 IDDR 端口 Q0/Q1/D/CLK/RST（Q0=上升沿，Q1=下降沿）；ODDR 端口 Q0/Q1/D0/D1/CLK/TX（D0=上升沿，D1=下降沿，TX 为第二沿时钟）。

4.1 RX：IDDR 采样 + 前导码/SOF 检测

module rgmii_rx (
    input  wire        rst_n,
    input  wire        rxc,
    input  wire [3:0]  rxd,
    input  wire        rx_ctl,
    // 输出 8bit GMII
    output wire [7:0]  rx_data,
    output wire        rx_dv,
    output wire        rx_sof       // 帧起始脉冲（SFD 后第一字节）
);
    wire [3:0] d_rise, d_fall;
    wire       ctl_rise;
    genvar i;
    generate
        for (i = 0; i < 4; i = i + 1) begin : g_rx
            IDDR #(.Q0_INIT(1'b0), .Q1_INIT(1'b0)) u_iddr (
                .Q0  (d_rise[i]),  .Q1  (d_fall[i]),
                .D   (rxd[i]),     .CLK (rxc),      .RST (~rst_n));
        end
    endgenerate
    IDDR #(.Q0_INIT(1'b0), .Q1_INIT(1'b0)) u_iddr_ctl (
        .Q0 (ctl_rise), .Q1 (), .D (rx_ctl), .CLK (rxc), .RST (~rst_n));

    // 一个 RXC 周期内：上升沿取低 4bit，下降沿取高 4bit
    assign rx_data = {d_fall, d_rise};
    assign rx_dv   = ctl_rise;

    // SOF 检测：检测到 SFD(0xD5) 后的下一字节为帧首
    // 用移位寄存判断前导码 55 55 55 55 55 55 55 D5
    reg [63:0] shift;
    reg        in_preamble;
    always @(posedge rxc or negedge rst_n)
        if (!rst_n) begin shift <= 0; in_preamble <= 1; end
        else if (rx_dv) begin
            shift <= {shift[55:0], rx_data};
            if (shift[63:8] == 56'h55555555555555 && shift[7:0] == 8'hD5)
                in_preamble <= 0;
        end else
            in_preamble <= 1;
    assign rx_sof = rx_dv && (shift[63:8] == 56'h55555555555555) && (shift[7:0] == 8'hD5);
endmodule

说明：RGMII 的 RX_CTL 在帧期间为高，前导码 + SFD 也随数据一起进来。rx_sof 标记"前导码结束、目的 MAC 首字节"这一拍，供 mac_rx 使用。若 PHY 已开 RX delay，直接在 RXC 边沿采样即可，无需再补延迟。

4.2 TX：ODDR 输出 + TXC 90° 相移

module rgmii_tx (
    input  wire        rst_n,
    input  wire        clk_125m,      // 0°
    input  wire        clk_125m_90,   // 90°（给 TXC）
    input  wire [7:0]  tx_data,
    input  wire        tx_en,
    // RGMII 输出
    output wire        txc,
    output wire [3:0]  txd,
    output wire        tx_ctl
);
    genvar i;
    generate
        for (i = 0; i < 4; i = i + 1) begin : g_tx
            ODDR #(.INIT(1'b0), .TXCLK_POL(1'b0)) u_oddr (
                .Q0 (txd[i]), .Q1 (),
                .D0 (tx_data[i]),   // 上升沿 = 低 4bit
                .D1 (tx_data[i+4]), // 下降沿 = 高 4bit
                .CLK (clk_125m), .TX (clk_125m));
        end
    endgenerate
    ODDR #(.INIT(1'b0), .TXCLK_POL(1'b0)) u_oddr_ctl (
        .Q0 (tx_ctl), .Q1 (), .D0 (tx_en), .D1 (tx_en),
        .CLK (clk_125m), .TX (clk_125m));
    // TXC：用 90° 相移时钟让 ODDR 恒翻转（D0=0, D1=1），得到滞后数据 2ns 的时钟
    ODDR #(.INIT(1'b0), .TXCLK_POL(1'b0)) u_oddr_txc (
        .Q0 (txc), .Q1 (), .D0 (1'b0), .D1 (1'b1),
        .CLK (clk_125m_90), .TX (clk_125m_90));
endmodule

2ns 偏斜的两种选择（二选一）：① FPGA 用 90° 相移时钟输出 TXC（如上）；② 打开 PHY 内部 TX delay，FPGA 直接用 0° 时钟。两者都开会导致 4ns 偏移、反而出错。P1-S3 确认 PHY 型号后定一种，写进 pins.md。

5. crc32：以太网 CRC 引擎

以太网 FCS 用 CRC-32，反射多项式 0xEDB88320，初值 0xFFFFFFFF，LSB 先处理。不要用非反射的 0x04C11DB7（那是另一套位序）。

5.1 完整实现（8bit/拍）

module crc32 (
    input  wire        clk, rst_n,
    input  wire        crc_en,     // 当前字节参与
    input  wire        crc_clr,    // 新帧复位（回到 0xFFFFFFFF）
    input  wire [7:0]  data,
    output wire [31:0] crc         // 寄存器值（未取反）
);
    reg [31:0] crc_reg;
    integer k;
    always @(posedge clk or negedge rst_n) begin
        if (!rst_n)       crc_reg <= 32'hFFFFFFFF;
        else if (crc_clr) crc_reg <= 32'hFFFFFFFF;
        else if (crc_en)  crc_reg <= next_crc(crc_reg, data);
    end
    function [31:0] next_crc;
        input [31:0] c; input [7:0] d;
        reg [31:0] t; integer j;
        begin
            t = c ^ {24'h00, d};
            for (j = 0; j < 8; j = j + 1)
                t = t[0] ? ((t >> 1) ^ 32'hEDB88320) : (t >> 1);
            next_crc = t;
        end
    endfunction
    assign crc = crc_reg;
endmodule

5.2 发送与校验约定（关键）

  • 发送 FCS：对"目的 MAC→payload"算完 CRC 后，寄存器值为 C，追加的 4 字节 FCS = ~C 的小端字节序：fcs[0]=~C[7:0], fcs[1]=~C[15:8], fcs[2]=~C[23:16], fcs[3]=~C[31:24]，各字节按 MSB-first 上线。
  • 接收校验：把整帧（含 FCS）喂入同一个 LFSR，最后一字节后寄存器应等于残差 0xC704DD7B，相等即 crc_ok。
  • 软件参考：binascii.crc32(payload) & 0xFFFFFFFF 的小端字节就是 FCS，直接用 Python 比对。

6.2 mac_tx：发送状态机

module mac_tx (
    input  wire        clk, rst_n,
    // 来自上层（eth_proto）的完整帧：目的MAC 起，不含前导码/FCS
    input  wire [7:0]  s_data,
    input  wire        s_valid,     // 电平
    input  wire        s_sof,
    input  wire        s_eof,
    output wire        s_ready,
    // 到 rgmii_tx
    output reg  [7:0]  m_data,
    output reg         m_valid,
    output reg         m_eof
);
    localparam IDLE=0, PREAMBLE=1, DATA=2, CRC=3, IFG=4;
    reg [2:0] state;
    reg [2:0] pre_cnt;      // 前导码计数 0..7
    reg [1:0] crc_cnt;      // FCS 4 字节
    reg [3:0] ifg_cnt;      // IFG ≥12 字节
    wire [31:0] crc_val;
    reg  crc_en, crc_clr;
    reg [31:0] fcs;

    crc32 u_crc (.clk(clk), .rst_n(rst_n), .crc_en(crc_en),
                 .crc_clr(crc_clr), .data(m_data), .crc(crc_val));

    assign s_ready = (state == DATA);

    always @(posedge clk or negedge rst_n) begin
        if (!rst_n) begin
            state <= IDLE; pre_cnt <= 0; crc_cnt <= 0; ifg_cnt <= 0;
            m_valid <= 0; m_data <= 0; crc_en <= 0; crc_clr <= 1;
        end else begin
            crc_clr <= 0; m_eof <= 0;
            case (state)
            IDLE: if (s_valid && s_sof) begin
                    state <= PREAMBLE; pre_cnt <= 0; crc_clr <= 1;
                  end
            PREAMBLE: begin
                    m_valid <= 1;
                    m_data <= (pre_cnt == 7) ? 8'hD5 : 8'h55;  // 55..55 D5
                    pre_cnt <= pre_cnt + 1;
                    if (pre_cnt == 7) state <= DATA;
                  end
            DATA: begin
                    if (s_valid) begin
                        m_data <= s_data; crc_en <= 1;
                        if (s_eof) begin state <= CRC; crc_cnt <= 0;
                            fcs <= ~crc_val; crc_en <= 0; end
                    end else crc_en <= 0;
                  end
            CRC: begin
                    m_valid <= 1;
                    case (crc_cnt)
                    2'd0: m_data <= fcs[7:0];
                    2'd1: m_data <= fcs[15:8];
                    2'd2: m_data <= fcs[23:16];
                    2'd3: m_data <= fcs[31:24];
                    endcase
                    crc_cnt <= crc_cnt + 1;
                    if (crc_cnt == 3) begin state <= IFG; ifg_cnt <= 0; m_valid <= 0; end
                  end
            IFG: begin
                    ifg_cnt <= ifg_cnt + 1;
                    if (ifg_cnt == 12) state <= IDLE;   // ≥96bit
                  end
            endcase
        end
    end
endmodule

注意：fcs <= ~crc_val 取的是"DATA 最后一拍前一拍"的寄存器值，因为 crc_en 与数据同拍有效、CRC 下一拍才更新。这里用组合 crc_val 在 s_eof 拍锁存，实际是"最后一拍数据已计入后的值"，上板前用 Python 参考对一帧确认（见 §10）。

6.3 mac_rx：接收状态机 + 地址过滤 + CRC 校验

module mac_rx #(
    parameter [47:0] LOCAL_MAC = 48'h02_00_00_00_00_01
)(
    input  wire        clk, rst_n,
    // 来自 rgmii_rx
    input  wire [7:0]  rx_data,
    input  wire        rx_dv,
    input  wire        rx_sof,
    // 到 eth_proto（payload：目的MAC 起，不含 FCS）
    output reg  [7:0]  m_data,
    output reg         m_valid,
    output reg         m_sof,
    output reg         m_eof,
    output reg         m_crc_ok     // eof 拍有效
);
    localparam IDLE=0, HDR=1, PAYLOAD=2, DROP=3;
    reg [1:0] state;
    reg [10:0] cnt;
    reg [47:0] dst_mac;
    reg        broadcast;
    wire [31:0] crc_val;
    reg  crc_en, crc_clr;
    reg [31:0] fcs_shift;

    crc32 u_crc (.clk(clk), .rst_n(rst_n), .crc_en(crc_en),
                 .crc_clr(crc_clr), .data(rx_data), .crc(crc_val));

    always @(posedge clk or negedge rst_n) begin
        if (!rst_n) begin
            state <= IDLE; cnt <= 0; dst_mac <= 0; broadcast <= 0;
            m_valid <= 0; m_sof <= 0; m_eof <= 0; m_crc_ok <= 0;
            crc_en <= 0; crc_clr <= 1;
        end else begin
            m_sof <= 0; m_eof <= 0; crc_clr <= 0;
            if (rx_sof) begin          // 帧首，复位各计数
                state <= HDR; cnt <= 0; crc_clr <= 1; crc_en <= 1;
            end else if (rx_dv) begin
                crc_en <= 1;
                case (state)
                HDR: begin
                        dst_mac <= {dst_mac[39:0], rx_data};
                        cnt <= cnt + 1;
                        if (cnt == 5) begin
                            broadcast <= (dst_mac == 48'hFFFF_FFFF_FFFF);
                            if (dst_mac == LOCAL_MAC || broadcast)
                                begin state <= PAYLOAD; m_valid <= 1; m_sof <= 1; end
                            else state <= DROP;     // 目的地址不匹配，丢
                        end else begin
                            m_data <= rx_data; m_valid <= 1;
                        end
                      end
                PAYLOAD: begin
                        m_data <= rx_data; m_valid <= 1;
                      end
                endcase
            end else if (state == PAYLOAD) begin
                // rx_dv 拉低：帧结束，最后 4 字节是 FCS
                crc_en <= 0; m_valid <= 0; m_eof <= 1;
                m_crc_ok <= (crc_val == 32'hC704DD7B);   // 残差校验
                state <= IDLE;
            end else if (state == DROP) begin
                state <= IDLE; crc_en <= 0;
            end
        end
    end
endmodule

要点：目的 MAC 在 HDR 阶段用移位寄存器收集 6 字节后判断；FCS 在 rx_dv 拉低当拍，其前 4 字节已计入 CRC，直接比对残差 0xC704DD7B。注意 m_data 在 DROP 时不输出，eth_proto 看到的是完整干净帧。

7. eth_proto：ARP / ICMP / UDP 回显

让 ping 通 + iperf -u 跑的最小协议栈。采用存转发（store-and-forward）：收帧存入 RAM 并顺带解析关键字段 → 帧校验通过后决定应答类型 → 组包写入发送 RAM → 校验和回填 → 交给 mac_tx。

7.1 应答帧布局（发送 RAM 字节偏移）

偏移ARP 应答ICMP 应答UDP 回显
0–5目的 MAC = 收帧源 MAC
6–11源 MAC = LOCAL_MAC
12–130x08060x08000x0800
14+ARP 体(28B)IP头(20B)+ICMP(8B)+payloadIP头(20B)+UDP(8B)+payload

7.2 完整实现（含 IP/ICMP 校验和）

module eth_proto #(
    parameter [47:0] LOCAL_MAC = 48'h02_00_00_00_00_01,
    parameter [31:0] LOCAL_IP  = {8'd192, 8'd168, 8'd1, 8'd100},
    parameter [15:0] ECHO_PORT = 16'd50000
)(
    input  wire        clk, rst_n,
    // RX（来自 mac_rx）
    input  wire [7:0]  rx_data,  input wire rx_valid,
    input  wire        rx_sof,   input wire rx_eof,  input wire rx_crc_ok,
    // TX（到 mac_tx）
    output reg  [7:0]  tx_data,  output reg tx_valid,
    output reg         tx_sof,   output reg tx_eof,  input wire tx_ready
);
    // ---- 帧缓冲（推断为 BSRAM）----
    reg [7:0]  rx_mem [0:2047];
    reg [10:0] rx_len;              // 收帧字节数
    // ---- 解析出的字段 ----
    reg [47:0] p_src_mac;  reg [31:0] p_src_ip;  reg [15:0] p_src_port;
    reg [15:0] p_ether;    reg [7:0]  p_proto;   reg [15:0] p_icmp_id, p_icmp_seq, p_ip_id;
    reg        p_is_arp, p_is_icmp, p_is_udp, p_ok;

    localparam S_IDLE=0, S_RX=1, S_DECIDE=2, S_BUILD=3,
               S_CSUM_IP=4, S_CSUM_ICMP=5, S_SEND=6;
    reg [2:0] state;
    reg [10:0] ptr;                 // 通用指针
    reg [15:0] sum;                 // 校验和累加
    reg [10:0] tx_len;

    // ============ 主状态机 ============
    always @(posedge clk or negedge rst_n) begin
        if (!rst_n) begin
            state <= S_IDLE; ptr <= 0; tx_valid <= 0; tx_sof <= 0; tx_eof <= 0;
        end else begin
            tx_sof <= 0; tx_eof <= 0;
            case (state)

            // ---- 收帧：写入 RAM + 按字节偏移解析 ----
            S_IDLE: if (rx_sof) begin state <= S_RX; ptr <= 0; rx_len <= 0;
                          p_ether <= 0; p_ok <= 0; end
            S_RX: if (rx_valid) begin
                    rx_mem[ptr] <= rx_data;
                    case (ptr)
                      6:  p_src_mac[47:40] <= rx_data;
                      7:  p_src_mac[39:32] <= rx_data;
                      8:  p_src_mac[31:24] <= rx_data;
                      9:  p_src_mac[23:16] <= rx_data;
                      10: p_src_mac[15:8]  <= rx_data;
                      11: p_src_mac[7:0]   <= rx_data;
                      12: p_ether[15:8]    <= rx_data;
                      13: p_ether[7:0]     <= rx_data;
                      26: p_src_ip[31:24]  <= rx_data;  // IP 头内
                      27: p_src_ip[23:16]  <= rx_data;
                      28: p_src_ip[15:8]   <= rx_data;
                      29: p_src_ip[7:0]    <= rx_data;
                      30: p_src_port[15:8] <= rx_data;  // UDP 源端口
                      31: p_src_port[7:0]  <= rx_data;
                      34: p_icmp_id[15:8]  <= rx_data;
                      35: p_icmp_id[7:0]   <= rx_data;
                      36: p_icmp_seq[15:8] <= rx_data;
                      37: p_icmp_seq[7:0]  <= rx_data;
                      18: p_ip_id[15:8]    <= rx_data;
                      19: p_ip_id[7:0]     <= rx_data;
                      23: p_proto          <= rx_data;  // IP 协议号
                    endcase
                    ptr <= ptr + 1; rx_len <= rx_len + 1;
                    if (rx_eof) state <= S_DECIDE;
                  end

            // ---- 决策应答类型 ----
            S_DECIDE: begin
                    p_is_arp  <= (p_ether == 16'h0806);
                    p_is_icmp <= (p_ether == 16'h0800) && (p_proto == 8'd1);
                    p_is_udp  <= (p_ether == 16'h0800) && (p_proto == 8'd17)
                                 && (p_src_port == ECHO_PORT);
                    p_ok <= rx_crc_ok && (p_is_arp || p_is_icmp || p_is_udp);
                    ptr <= 0;
                    state <= p_ok ? S_BUILD : S_IDLE;
                  end

            // ---- 组包（详见 7.3 的 write_byte 逻辑，此处抽象）----
            S_BUILD: begin
                    // 按 ptr 顺序写 tx_mem：MAC头 + 对应协议体 + payload 回显
                    if (ptr == tx_len) begin
                        state <= (p_is_arp) ? S_SEND : S_CSUM_IP; ptr <= 0;
                    end else ptr <= ptr + 1;
                  end

            // ---- IP 头校验和（20B，字段[24..25]置 0 后求反）----
            S_CSUM_IP: begin
                    // 累加 tx_mem[14..33] 的 16bit 大端字，写回 [24..25]
                    state <= (p_is_icmp) ? S_CSUM_ICMP : S_SEND;
                  end

            // ---- ICMP 校验和（type..payload）----
            S_CSUM_ICMP: begin
                    state <= S_SEND;
                  end

            // ---- 发送给 mac_tx ----
            S_SEND: begin
                    if (tx_ready) begin
                        tx_data <= rx_mem[ptr]; // 复用 RAM 或独立 tx_mem
                        tx_valid <= 1;
                        tx_sof <= (ptr == 0);
                        ptr <= ptr + 1;
                        if (ptr == tx_len) begin tx_eof <= 1; tx_valid <= 0; state <= S_IDLE; end
                    end
                  end
            endcase
        end
    end
endmodule

7.3 组包细节（BUILD 阶段按指针写出）

为避免单模块代码过长，组包逻辑用"按 ptr 查表"实现。核心规则：

  • MAC 头：[0..5]=p_src_mac，[6..11]=LOCAL_MAC，[12..13]=0x0806 或 0x0800。
  • ARP 应答（28B，tx_len=42）：htype=1, ptype=0x0800, hlen=6, plen=4, op=2，sender=LOCAL_MAC/IP，target=p_src_mac/IP。
  • ICMP 应答（tx_len=42+payload）：IP 头（ver=0x45, TTL=64, proto=1, src=LOCAL_IP, dst=p_src_ip），ICMP 头（type=0 echo reply, code=0, id/seq 回填），payload 原样回显。
  • UDP 回显（tx_len=42+payload）：IP 头（proto=17），UDP 头（src=ECHO_PORT, dst=p_src_port, 长度，checksum=0 表示不校验），payload 原样回显。

7.4 校验和算法（Internet checksum，16bit 反码和）

// 通用 16bit 校验和：把 [start..end] 按大端两字节求和，折回进位，取反
function [15:0] cksum;
    input [15:0] acc;      // 当前累加值
    input [7:0]  hi, lo;   // 一个 16bit 字的高/低字节
    reg [16:0] s;
    begin
        s = {1'b0, acc} + {1'b0, hi, lo};
        if (s[16]) s = s + 1;      // 进位折回
        cksum = s[15:0];
    end
endfunction
// 最终字段值 = ~cksum（反码）。IP 头校验和只覆盖 20B 头；ICMP 覆盖 type 到 payload 末尾。

UDP 校验和：IPv4 允许 UDP 校验和为 0（表示不校验）。v0.1 直接填 0 省去 UDP 伪首部计算；若 iperf -u 对端开启校验（Linux 默认开启但 0 视为有效），即可通过。

8. fifo_cdc：跨时钟缓冲（P3 预留）

v0.1 全部在 125MHz 域，用旁路直连；但接口按异步 FIFO 设计，P3 接入 61.44MHz 基带时只换实现。

8.1 异步 FIFO 设计（格雷码指针同步）

  • 双口 RAM 深度取 2 的幂（如 2048），写/读指针各带 格雷码转换。
  • 读指针格雷码经两级寄存器同步到写域算 full；写指针同步到读域算 empty。
  • 判断：empty = (rd_ptr_gray == wr_ptr_sync)；full = (wr_ptr_gray == ~rd_ptr_sync[高2位], rd_ptr_sync[低位])。
// 格雷码转换与两级同步（示意）
assign wgray = wptr ^ (wptr >> 1);
always @(posedge rclk) begin wgray_s1 <= wgray; wgray_s2 <= wgray_s1; end
assign rempty = (rgray == wgray_s2);
// 空/满判断后，用格雷码转回二进制作为 RAM 地址

P3 落点：到 P3 时，eth_proto 的 TX/RX 侧分别换成 FIFO 的读/写端口，上层协议逻辑一行不改，即可把 PC 数据接到 frame_tx/frame_rx。

# gen_frames.py —— 生成激励帧与期望应答，供 testbench 比对
import binascii, struct

LOCAL_MAC = bytes([0x02,0,0,0,0,1])
LOCAL_IP  = (192,168,1,100)
PC_MAC    = bytes([0x00,0x1A,0x2B,0x3C,0x4D,0x5E])
PC_IP     = (192,168,1,10)

def eth_fcs(payload: bytes) -> bytes:
    # binascii.crc32 即以太网 CRC，小端 4 字节为 FCS
    return binascii.crc32(payload).to_bytes(4, 'little')

def cksum(data: bytes) -> int:
    if len(data) % 2: data += b'\x00'
    s = sum(struct.unpack('!%dH' % (len(data)//2), data))
    while s >> 16: s = (s & 0xffff) + (s >> 16)
    return (~s) & 0xffff
